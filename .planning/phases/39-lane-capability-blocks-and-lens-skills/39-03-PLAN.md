```python
import json
import unicodedata
from pathlib import Path

REPO = Path(__file__).resolve().parents[3]
# SPEC R8: Cc (U+0000-001F, U+007F-009F), Zl (U+2028), Zp (U+2029) and the listed format points.
REJECTED = [(0x0000, 0x001F), (0x007F, 0x009F), (0x200B, 0x200F), (0x2028, 0x2029),
            (0x202A, 0x202E), (0x2060, 0x2064), (0x2066, 0x2069), (0xFEFF, 0xFEFF)]
ACCEPTED = [0x0020, 0x007E, 0x00A0, 0x00AD, 0x00E9, 0x200A, 0x2010, 0x2027, 0x202F, 0x205F,
            0x2065, 0x206A, 0xFEFE, 0xFFFD, 0x1F600]


def entry(cp):
    fallback = "<control>" if cp <= 0x9F else "<unassigned>"
    return json.dumps([f"{cp:04X}", unicodedata.name(chr(cp), fallback)])


def block(points):
    return ",\n".join(f"    {entry(cp)}" for cp in points)


rejected = [cp for low, high in REJECTED for cp in range(low, high + 1)]
text = ('{\n  "rejected": [\n' + block(rejected) + '\n  ],\n  "accepted": [\n'
        + block(ACCEPTED) + "\n  ]\n}\n")
(REPO / "tests" / "fixtures" / "prompt_control_chars.json").write_text(text, encoding="utf-8")
```
