"""38-08 Task 2: move the self-test harness from cross-ai.mjs into cross-ai-selftest.mjs.

Run from the checkout root: python3 tmp/run-2026-11-19/p38-08/splice.py
Cuts four blocks by unique markers, derives one import list (sibling) and one export list
(runner), replaces the self-test entry in main with a loader, and builds the candidate pair

Run it once; every later check is read-only.

import re
import shutil
import subprocess
import sys
from pathlib import Path

T = Path("skills/ai-kit-spec-review/references/cross-ai.mjs")
S = T.with_name("cross-ai-selftest.mjs")
SCRATCH = Path("tmp/run-2026-11-19/p38-08")
ORIG = SCRATCH / "cross-ai.mjs.orig"
# (start, end): a block runs from its start marker up to, not including, its end marker.
BLOCKS = [

     "// Flag, then an explicit CONFIG boolean, then the name heuristic, else unknown (D-05)."),
    ("function classifyOne(entry, table) {",
     "function which(bin, envPath = process.env.PATH || '') {"),
    ("// A case's state file, resolved inside the case's own XDG state dir.",
     "function readCycleState(file) {"),
    ("function hermeticEnv(tmp, extra = {}) {", "function callOf(o, name, model, newSessionId"),

ENTRY_OLD = "  if (o.selfTest) return blackBoxSelfTest(o.selfTestList);\n"
ENTRY_NEW = (
    "  if (o.selfTest) {\n"
    "    const harness = await loadSelfTest();\n"
    "    return harness ? harness(o.selfTestList) : 2;\n"
    "  }\n"

MAIN_MARK = "async function main(argv) {\n"
LOADER = (

    "// sibling and refuses; a sibling that exists but fails to load is a crash, not a refusal.\n"
    "const SELF_TEST_FILE = 'cross-ai-selftest.mjs';\n"
    "async function loadSelfTest() {\n"
    "  const file = path.join(path.dirname(RUNNER_FILE), SELF_TEST_FILE);\n"
    "  let mod = null;\n"
    "  try {\n"
    "    fs.accessSync(file, fs.constants.R_OK);\n"
    "    const st = fs.statSync(file);\n"
    "    if (st.isFile() && st.size > 0) mod = await import(new URL(`./${SELF_TEST_FILE}`, import.meta.url));\n"
    "  } catch (e) {\n"
    "    // Only a file that is gone or unreadable refuses; a syntax or link error stays a crash.\n"
    "    if (!['ENOENT', 'EACCES', 'EPERM', 'EISDIR', 'ERR_MODULE_NOT_FOUND'].includes(e && e.code)) throw e;\n"
    "  }\n"
    "  if (mod && typeof mod.blackBoxSelfTest === 'function') return mod.blackBoxSelfTest;\n"
    "  log('cross-ai: the self-test is not shipped in this copy; run it from the template');\n"
    "  return null;\n"
    "}\n\n"
)
HEADER = (
    "// ai-kit cross-ai self-test harness (38-08). cross-ai.mjs loads this file only for\n"
    "// --self-test and --self-test-list; a rendered project copy does not carry it.\n"
)
EXPORT_NOTE = (
    "// 38-08: the names the self-test harness (cross-ai-selftest.mjs) imports. One list,\n"
    "// never per-declaration exports (the test extractors read `function name(` lines).\n"
)
GONE = ["function codexPair(", "const CLASSIFY_CASES = [", "function classifyCases(",
        "function caseCycleFile(", "function hermeticEnv(", "function writeCopy(",
        "const FAKE_PRELUDE = String.raw`", "async function blackBoxSelfTest(",
        "selfTestRoot", "selfTestAbort"]
KEPT = ["function pidAlive(", "function procStartTime(", "function procGroup(",
        "function procSession(", "function groupMembers(", "function routeOf(",
        "function confirmedRoute(", "function setTable(value)", "function setOverlayIds(value)",
        "function setLog(fn)", "function setSelfTestHook(fn)", "const RUNNER_FILE = "]
# SPEC R8 (amended 2026-11-21): the liveness helpers stay in the runner and are exported.
REQUIRED = {"pidAlive", "procStartTime", "procGroup", "procSession", "groupMembers"}
TOP = re.compile(
    r"^(?:async\s+)?function\*?\s+([A-Za-z_$][\w$]*)|^(?:const|let|var|class)\s+([A-Za-z_$][\w$]*)",
    re.M)
DESTRUCT = re.compile(r"^(?:const|let|var)\s*[\[{]", re.M)
REBIND = re.compile(r"^\s*(?:TABLE|OVERLAY_IDS|log)\s*=(?!=)", re.M)


def run_env() -> dict:
    run = SCRATCH / "run"
    shutil.rmtree(run, ignore_errors=True)
    for sub in ("home", "state", "tmp"):
        (run / sub).mkdir(parents=True)
    env = dict(os.environ)
    env.update(HOME=str((run / "home").resolve()), XDG_STATE_HOME=str((run / "state").resolve()),
               TMPDIR=str((run / "tmp").resolve()), AI_KIT_REFUSE_REAL_HOME="1")
    return env


def run_node(argv: list[str], timeout: int) -> subprocess.CompletedProcess:
    # Own session, whole group killed on timeout: the PATH node is a Volta shim (detection.py:951).
    proc = subprocess.Popen(argv, stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                            env=run_env(), start_new_session=True)
    try:
        out, err = proc.communicate(timeout=timeout)
    except subprocess.TimeoutExpired:
        os.killpg(proc.pid, 9)
        out, err = proc.communicate()
        return subprocess.CompletedProcess(argv, 124, out, err + b"\nsplice: timed out")
    return subprocess.CompletedProcess(argv, proc.returncode, out, err)


def node_bin() -> str:
    proc = run_node(["node", "-e", "process.stdout.write(process.execPath)"], 60)
    path = proc.stdout.decode("utf-8", "replace")
    if proc.returncode != 0 or not os.path.isfile(path):
        sys.exit("splice: no node executable found")
    return path


def listing(node: str, runner: Path) -> bytes | None:
    proc = run_node([node, str(runner), "--self-test-list"], 300)
    return proc.stdout if proc.returncode == 0 and proc.stdout else None


def top_names(text: str) -> set[str]:
    return {a or b for a, b in TOP.findall(text)}


def cut(text: str) -> tuple[str, str]:
    spans = []
    for start_mark, end_mark in BLOCKS:
        for mark in (start_mark, end_mark):
            if text.count(mark) != 1:
                sys.exit(f"splice: marker occurs {text.count(mark)} times, expected 1: {mark[:60]!r}")
        start, end = text.index(start_mark), text.index(end_mark)
        if end <= start:
            sys.exit(f"splice: block out of order: {start_mark[:60]!r}")
        spans.append((start, end))
    if any(a[1] > b[0] for a, b in zip(spans, spans[1:])):
        sys.exit("splice: blocks overlap or are out of order")
    body = "".join(text[s:e] for s, e in spans)
    runner = text
    for s, e in reversed(spans):
        runner = runner[:s] + runner[e:]
    return runner, body


def build(text: str, exclude: set[str]) -> tuple[str, str, list[str]]:
    runner, body = cut(text)
    if DESTRUCT.search(runner) or DESTRUCT.search(body):
        sys.exit("splice: a top-level destructuring declaration needs a manual census")
    if REBIND.search(body):
        sys.exit("splice: the harness still assigns TABLE, OVERLAY_IDS or log (Task 1 missing)")
    if "import.meta.url" in body:
        sys.exit("splice: the harness still reads import.meta.url (Task 1 missing)")
    if runner.count(ENTRY_OLD) != 1 or runner.count(MAIN_MARK) != 1:
        sys.exit("splice: the self-test entry or the first line of main is not unique")
    runner = runner.replace(ENTRY_OLD, ENTRY_NEW, 1).replace(MAIN_MARK, LOADER + MAIN_MARK, 1)
    # Not minus the harness's own names: the fake scripts hold column-0 lines such as
    # `const log = ...` inside strings, and a real harness declaration could never share a name
    # with a runner one (both lived in one module).
    used = sorted(REQUIRED | {name for name in top_names(runner) - exclude
                              if re.search(r"(?<![\w$])" + re.escape(name) + r"(?![\w$])", body)})
    imports = "".join(line + "\n" for line in text.split("// <config>", 1)[0].splitlines()
                      if line.startswith("import "))
    names = "".join(f"  {name},\n" for name in used)
    sibling = (HEADER + imports + "import {\n" + names + "} from './cross-ai.mjs';\n\n"
               + body.rstrip("\n") + "\n\nexport { blackBoxSelfTest };\n")
    runner = runner.rstrip("\n") + "\n\n" + EXPORT_NOTE + "export {\n" + names + "};\n"
    for mark in GONE:
        if mark in runner:
            sys.exit(f"splice: the runner still holds {mark!r}")
    for mark in KEPT:
        if mark not in runner:
            sys.exit(f"splice: the runner lost {mark!r}")
    return runner, sibling, used


def node_check(node: str, path: Path) -> str | None:
    proc = run_node([node, "--check", str(path)], 120)
    err = proc.stderr.decode("utf-8", "replace")[-1500:]
    return None if proc.returncode == 0 else f"node --check {path}: {err}"


DROP_RX = (re.compile(r"Export '([\w$]+)' is not defined"),
           re.compile(r"Identifier '([\w$]+)' has already been declared"))


def main() -> int:
    if S.exists():
        sys.exit(f"splice: {S} already exists (the splice already ran; do not run it again)")
    cand = SCRATCH / "cand"
    shutil.rmtree(cand, ignore_errors=True)
    cand.mkdir(parents=True)
    cand_t, cand_s = cand / T.name, cand / S.name
    text = T.read_text(encoding="utf-8")
    exclude: set[str] = set()
    runner, sibling, used = "", "", []
    try:
        node = node_bin()
        before = listing(node, T)
        if before is None:
            sys.exit("splice: --self-test-list failed before the cut; nothing written")
        shutil.copyfile(T, ORIG)
        (SCRATCH / "list-before.txt").write_bytes(before)
        problems: list[str] = []
        for _attempt in range(25):
            runner, sibling, used = build(text, exclude)
            cand_t.write_text(runner, encoding="utf-8")
            cand_s.write_text(sibling, encoding="utf-8")
            problems = [p for p in (node_check(node, cand_t), node_check(node, cand_s)) if p]
            # A column-0 declaration inside a runner string looks like a top-level name; node
            # names it (an undeclared export, or a clash with a harness one): drop, rebuild.
            dropped = {m.group(1) for p in problems for rx in DROP_RX for m in rx.finditer(p)}
            if not dropped - exclude:
                break
            exclude |= dropped
        after = None if problems else listing(node, cand_t)
        if not problems and after != before:
            (SCRATCH / "list-after.txt").write_bytes(after or b"")
            problems.append("--self-test-list differs after the cut (see list-after.txt)")
        if problems:
            print("splice: " + "; ".join(problems) + "; nothing outside tmp/ changed")
            return 1
        # Publish only the validated pair; skills/ is written by these two copies and nothing else.
        shutil.copyfile(cand_s, S)
        shutil.copyfile(cand_t, T)
    finally:
        shutil.rmtree(SCRATCH / "run", ignore_errors=True)
    print(f"splice: ok, runner {len(runner.splitlines())} lines, "
          f"sibling {len(sibling.splitlines())} lines, {len(used)} shared names")
    print("splice: shared names: " + " ".join(used))
    print("splice: dropped string-only names: " + (" ".join(sorted(exclude)) or "-"))
    return 0


if __name__ == "__main__":
    sys.exit(main())
