import json
import sys

ROOT = Path(__file__).resolve().parents[3]
CANONICAL = ROOT / "skills" / "ai-kit-spec-review" / "ai_kit_spec" / "secret_shapes.py"
TEMPLATE = ROOT / "skills" / "ai-kit-spec-review" / "references" / "cross-ai.mjs"
OLD = (
    "const SECRET_RES = [\n"
    "  /(sk|rk)-[A-Za-z0-9_-]{8,}/gi,\n"
    "  /(?<=\\bBearer )[A-Za-z0-9._~+/=-]{8,}/gi,\n"
    "  /(?<=\\bBasic )[A-Za-z0-9+/=]{8,}/gi,\n"
    "  /\\bAIza[0-9A-Za-z_-]{20,}/g,\n"
    "  /conn:[0-9a-f]{8,}/gi,\n"
    "  /\\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\\b/gi,\n"
    "  /\\bgh[pousr]_[A-Za-z0-9]{20,}\\b/gi,\n"
    "  /\\bgithub_pat_[A-Za-z0-9_]{20,}\\b/gi,\n"
    "];\n"
)
NEW_TAIL = (
    "const SECRET_RES = KEY_SHAPES.map(([, source]) => new RegExp(source, 'g'));\n"
    "// Bearer and Basic values go before the key shapes, so a key-shaped run inside one cannot\n"
    "// stop the rule early; the other broad rules go after them.\n"
    "const BEARER_BASIC_RES = [\n"
    "  /(?<=\\bBearer )[A-Za-z0-9._~+/=-]{8,}/gi,\n"
    "  /(?<=\\bBasic )[A-Za-z0-9+/=]{8,}/gi,\n"
    "];\n"
    "const BROAD_RES = [\n"
    "  /conn:[0-9a-f]{8,}/gi,\n"
    "  /\\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\\b/gi,\n"
    "];\n"
    "\n"
    "// Every key shape is matched on the same text and overlapping matches merge into one span,\n"
    "// so no row cuts into a token another row matches; text outside the spans is kept.\n"
    "function redactKeyShapes(text) {\n"
    "  const spans = SECRET_RES.flatMap((re) => [...text.matchAll(re)])\n"
    "    .map((m) => [m.index, m.index + m[0].length])\n"
    "    .sort((a, b) => a[0] - b[0] || a[1] - b[1]);\n"
    "  const merged = [];\n"
    "  for (const [start, end] of spans) {\n"
    "    const lastEnd = merged.length ? merged[merged.length - 1][1] : -1;\n"
    "    if (start < lastEnd) merged[merged.length - 1][1] = Math.max(lastEnd, end);\n"
    "    else merged.push([start, end]);\n"
    "  }\n"
    "  let out = '';\n"
    "  let pos = 0;\n"
    "  for (const [start, end] of merged) {\n"
    "    out += text.slice(pos, start) + '<redacted>';\n"
    "    pos = end;\n"
    "  }\n"
    "  return out + text.slice(pos);\n"
    "}\n"
    "\n"
)
COMMENT = "// Distinctive key shapes as one JSON line, kept equal to the shared Python table by a test.\n"
OLD_LOOP = (
    "  for (const re of SECRET_RES) {\n"
    "    re.lastIndex = 0;\n"
    "    out = out.replace(re, '<redacted>');\n"
    "  }\n"
    "  return replaceCopies(out, copies);\n"
)
NEW_LOOP = (
    "  for (const re of BEARER_BASIC_RES) {\n"
    "    re.lastIndex = 0;\n"
    "    out = out.replace(re, '<redacted>');\n"
    "  }\n"
    "  out = redactKeyShapes(out);\n"
    "  for (const re of BROAD_RES) {\n"
    "    re.lastIndex = 0;\n"
    "    out = out.replace(re, '<redacted>');\n"
    "  }\n"
    "  return replaceCopies(out, copies);\n"
)


def canonical_shapes():
    tree = ast.parse(CANONICAL.read_text(encoding="utf-8"))
    for node in tree.body:
        targets = node.targets if isinstance(node, ast.Assign) else []
        if any(isinstance(t, ast.Name) and t.id == "SHAPES" for t in targets):
            return [list(row) for row in ast.literal_eval(node.value)]
    sys.exit("no SHAPES in the canonical file")


text = TEMPLATE.read_bytes().decode("utf-8")
for label, old in (("old SECRET_RES block", OLD), ("old redactSecrets loop", OLD_LOOP)):
    if text.count(old) != 1:
        sys.exit(f"{label} found {text.count(old)} times, expected 1")
shapes = canonical_shapes()
line = "const KEY_SHAPES = " + json.dumps(shapes, separators=(",", ":")) + ";\n"
text = text.replace(OLD, COMMENT + line + NEW_TAIL).replace(OLD_LOOP, NEW_LOOP)
TEMPLATE.write_bytes(text.encode("utf-8"))
print(f"spliced KEY_SHAPES ({len(shapes)} rows) into {TEMPLATE.relative_to(ROOT)}")
